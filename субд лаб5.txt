CREATE TABLE Employees(Employee_id INT IDENTITY(1,1) PRIMARY KEY,
		       First_Name NVARCHAR(30) NOT NULL,
		       Last_Name NVARCHAR(30) NOT NULL,
		       BirthDate DATE NOT NULL);

CREATE TABLE Categories(Category_id INT IDENTITY(1,1) PRIMARY KEY,
    			Category_Name NVARCHAR(30) NOT NULL);

CREATE TABLE Products(Product_id INT IDENTITY(1,1) PRIMARY KEY,
    		      Product_Name NVARCHAR(30) NOT NULL,
                      Category_id INT NOT NULL,
    		      Price money NOT NULL,
    		      CONSTRAINT FK_Products_Categories FOREIGN KEY (CategoryID) 
        	      REFERENCES Categories(CategoryID));

CREATE FUNCTION Choose_Category (@Category_Name NVARCHAR(30))
RETURNS TABLE
AS
RETURN
    (SELECT P.Product_id, P.Product_Name, P.Price FROM Products P
    INNER JOIN Categories C ON P.Category_id = C.Category_id
    WHERE C.Category_Name = @Category_Name);
Go
SELECT Product_id, Product_Name, Price FROM Choose_Category('Бакалея');

CREATE FUNCTION Form_Name (@First_Name NVARCHAR(30), @Last_Name NVARCHAR(30))
RETURNS NVARCHAR(60)
AS
BEGIN
	RETURN @First_Name + ' ' + @Last_Name
END;
Go
SELECT First_Name, Last_Name, dbo.Form_Name(First_Name, Last_Name) AS Full_Name FROM Employees;



Go
CREATE FUNCTION AvgPrice (@Category_id INT)
RETURNS INT
AS
BEGIN
    RETURN (SELECT ISNULL(AVG(Price), 0) 
            FROM Products 
            WHERE Category_id = @Category_id);
END;
Go
SELECT Category_Name, dbo.AvgPrice(Category_id) AS Average_Price FROM Categories;
